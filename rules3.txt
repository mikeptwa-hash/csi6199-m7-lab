rule silent_banker : PossibleAstronomer
{
    meta:
        description = "Detects astronomer, drake or biologist"
        threat_level = 5
        in_the_wild = true
    strings:
        $a = "astronomer"
        $b = "drake"
        $c = "Drake"
        $d = "biologist"
    condition:
        $a or $b or $c or $d
}
